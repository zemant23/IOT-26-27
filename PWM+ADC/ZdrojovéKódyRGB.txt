První kde byly špatně nastavené piny pro RGB:

from machine import Pin, PWM
from time import sleep

red = PWM(Pin(11))
green = PWM(Pin(12))
blue = PWM(Pin(13))

red.freq(1000)
green.freq(1000)
blue.freq(1000)

clk = Pin(18, Pin.IN, Pin.PULL_UP)
dt = Pin(19, Pin.IN, Pin.PULL_UP)
sw = Pin(20, Pin.IN, Pin.PULL_UP)

r = 255
g = 0
b = 0

barva = 0

def nastav_rgb():
    red.duty_u16(r * 257)
    green.duty_u16(g * 257)
    blue.duty_u16(b * 257)

def omez(hodnota):
    return max(0, min(255, hodnota))

nastav_rgb()

posledni_clk = clk.value()
posledni_sw = sw.value()

while True:
    aktualni_clk = clk.value()

    if aktualni_clk != posledni_clk:
        if dt.value() != aktualni_clk:
            smer = 1
        else:
            smer = -1

        if barva == 0:
            r = omez(r + smer)
        elif barva == 1:
            g = omez(g + smer)
        else:
            b = omez(b + smer)

        nastav_rgb()
        posledni_clk = aktualni_clk

    aktualni_sw = sw.value()

    if posledni_sw == 1 and aktualni_sw == 0:
        sleep(0.05)

        if sw.value() == 0:
            barva += 1

            if barva > 2:
                barva = 0

            while sw.value() == 0:
                sleep(0.01)

    posledni_sw = aktualni_sw

    sleep(0.001)
