
from machine import Pin, PWM, ADC
from time import sleep

red = PWM(Pin(13))
green = PWM(Pin(12))
blue = PWM(Pin(11))

red.freq(1000)
green.freq(1000)
blue.freq(1000)

pot = ADC(26)
sw = Pin(14, Pin.IN, Pin.PULL_UP)

r = 255
g = 0
b = 0

barva = 0

posledni_sw = 1
posledni_pot = -1

def nastav_rgb():
    red.duty_u16(r * 257)
    green.duty_u16(g * 257)
    blue.duty_u16(b * 257)

nastav_rgb()

while True:

    # Tlacitko - zmena vybrane barvy
    if sw.value() == 0 and posledni_sw == 1:
        sleep(0.05)

        if sw.value() == 0:
            barva = (barva + 1) % 3
            posledni_pot = -1

            if barva == 0:
                print("Nastavujes CERVENOU")
            elif barva == 1:
                print("Nastavujes ZELENOU")
            else:
                print("Nastavujes MODROU")

            while sw.value() == 0:
                sleep(0.01)

    posledni_sw = sw.value()

    # Potenciometr
    hodnota = pot.read_u16()

    # Zmena jen pri pohybu potenciometru
    if abs(hodnota - posledni_pot) > 300:

        jas = hodnota * 255 // 65535

        if barva == 0:
            r = jas
        elif barva == 1:
            g = jas
        else:
            b = jas

        nastav_rgb()
        posledni_pot = hodnota

    sleep(0.02)
